📊 DATABASE SCHEMA:

USERS TABLE:
  id (PK), name, email, password, created_at, updated_at

EXPENSES TABLE:
  id (PK), user_id (FK), title, category (ENUM), amount,
  payment_method (ENUM), notes, date, created_at, updated_at

INCOME TABLE:
  id (PK), user_id (FK), source, amount, date, notes, created_at

BUDGETS TABLE:
  id (PK), user_id (FK), monthly_budget, daily_limit,
  warn_threshold, updated_at
